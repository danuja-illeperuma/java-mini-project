/*Time Table*/

CREATE TABLE timetables (
    timetable_id INT AUTO_INCREMENT PRIMARY KEY,
    course_id INT NOT NULL,
    lecturer_id INT NOT NULL,
    day_of_week ENUM('Monday', 'Tuesday', 'Wednesday', 'Thursday', 'Friday', 'Saturday', 'Sunday') NOT NULL,
    start_time TIME NOT NULL,
    end_time TIME NOT NULL,
    location VARCHAR(50) NOT NULL,
    session_type ENUM('theory', 'practical') NOT NULL,
    FOREIGN KEY (course_id) REFERENCES courses(course_id),
    FOREIGN KEY (lecturer_id) REFERENCES users(user_id)
);

/* Normal attendance */
public void calculateAttendancePercentage(){

SELECT 
		username,
		c_code,
		c_type,
			
		SUM(CASE WHEN at_state = 'Present' THEN c_hours ELSE 0 END) * 100.0 /(SUM(c_hours)) 
		AS 
			Percentage
FROM 
	attendance_detail
WHERE 
	c_code = ?
	
GROUP BY 
	username,c_code,c_type;	
	
	
/* Medical with attendance for lecture */
public void AttendancePercentageWithMedical(){
	
	SELECT 
		username,
		c_code,
		c_type,
			
		SUM(CASE WHEN at_state = 'Present' OR at_state = 'Medical' THEN c_hours ELSE 0 END)* 100.0 /(SUM(c_hours)) 
		AS 
			Percentage,
		CASE 
			WHEN (SUM(CASE WHEN at_state = 'Present' OR at_state = 'Medical' THEN c_hours ELSE 0 END)* 100.0 /SUM(c_hours))  >= 80 
			THEN 
				'Eligible' 
			ELSE 
				'Not_Eligible' END  AS Eligibility		
			
FROM 
	attendance_detail
WHERE
	c_code = ?
	
GROUP BY 
	username,c_code,c_type;	

}

/*Final end marks calculate*/


CREATE VIEW final_end_marks AS
SELECT 
    student_id,
    course_code,
    CAST(
        CASE
            WHEN course_code IN ('ICT2113', 'ICT2133') 
                THEN (end_theory * 0.4 + end_practical * 0.3)
            WHEN course_code = 'ICT2122'
                THEN (end_theory * 0.6)
            WHEN course_code = 'ICT2142'
                THEN (end_practical * 0.6)
            WHEN course_code = 'ICT2152'
                THEN (end_theory * 0.7)
            ELSE NULL
        END 
    AS DECIMAL(4,2)) AS final_end_marks
FROM 
    end_marks;



/*Final CA marks calculate */


CREATE VIEW final_CA_marks AS
SELECT
    stu_id,
    c_code,
    CAST(
        CASE 
            WHEN c_code='ICT2113' THEN 
                (((Q1_mark + Q2_mark + Q3_mark) - LEAST(Q1_mark, Q2_mark, Q3_mark))/2) * 0.1 +
                (mid_term * 0.2)
            WHEN c_code = 'ICT2122' THEN 
                (((Q1_mark + Q2_mark + Q3_mark + Q4_mark) - LEAST(Q1_mark, Q2_mark, Q3_mark, Q4_mark))/3) * 0.1 +
                ((as1_mark + as2_mark) * 0.1) +
                (mid_term * 0.2)
	    WHEN c_code IN ('ICT2133','ICT2152') THEN 
                (((Q1_mark + Q2_mark + Q3_mark ) - LEAST(Q1_mark, Q2_mark, Q3_mark))/2) * 0.1 +
                ((as1_mark + as2_mark) * 0.2) 
	    WHEN c_code='ICT2142'  THEN
		((as1_mark + as2_mark) * 0.2) +
                (mid_term * 0.2)
            ELSE NULL
        END AS DECIMAL(4, 2)
    ) AS final_CA_marks
FROM 
    CA_marks;


/*Final marks calculate  (CA marks+final end marks)
View for final marks as a whole batch with all course units*/


CREATE VIEW final_results AS
SELECT 
    f.student_id,
    f.course_code,
    (f.final_end_marks + t.final_CA_marks) AS total_marks
FROM 
    final_end_marks f
JOIN 
    final_ca_marks t 
ON 
    f.student_id = t.stu_id AND f.course_code = t.c_code;



/* view for student grades  */

CREATE VIEW student_grades AS
SELECT
    fr.student_id,
    fr.course_code,
    
    CASE
        WHEN fr.total_marks >= 85.0 THEN 'A+'
        WHEN fr.total_marks >= 75.0 THEN 'A'
        WHEN fr.total_marks >= 70.0 THEN 'A-'
        WHEN fr.total_marks >= 65.0 THEN 'B+'
        WHEN fr.total_marks >= 60.0 THEN 'B'
        WHEN fr.total_marks >= 55.0 THEN 'B-'
        WHEN fr.total_marks >= 50.0 THEN 'C+'
        WHEN fr.total_marks >= 45.0 THEN 'C'
        WHEN fr.total_marks >= 40.0 THEN 'C-'
        WHEN fr.total_marks >= 35.0 THEN 'D+'
        WHEN fr.total_marks >= 30.0 THEN 'D'
        ELSE 'E'
    END AS student_grade
FROM
    final_results fr
JOIN
    users u ON fr.student_id = u.username;



 /* View course_credits table */

CREATE VIEW course_credits AS
SELECT
    s.student_id,
    c.course_code,
    c.credits,
    s.student_grade
FROM
    student_grades s
JOIN
    courses c ON s.course_code = c.course_code;

        WHEN student_grade = 'C+' THEN 2.3
        WHEN student_grade = 'C' THEN 2.0
        WHEN student_grade = 'C-' THEN 1.7
        WHEN student_grade = 'D+' THEN 1.3
        WHEN student_grade = 'D' THEN 1.0
        WHEN student_grade = 'E' THEN 0.0
        ELSE 0.0
    END) / SUM(credits), 2) AS SGPA
FROM course_credits
GROUP BY student_id;


/* Calculate CGPA */

CREATE VIEW CGPA AS
SELECT
    student_id,
    ROUND(SUM(credits * CASE
        WHEN student_grade = 'A+' THEN 4.0
        WHEN student_grade = 'A' THEN 4.0
        WHEN student_grade = 'A-' THEN 3.7
        WHEN student_grade = 'B+' THEN 3.3
        WHEN student_grade = 'B' THEN 3.0
        WHEN student_grade = 'B-' THEN 2.7
        WHEN student_grade = 'C+' THEN 2.3
        WHEN student_grade = 'C' THEN 2.0
        WHEN student_grade = 'C-' THEN 1.7
        WHEN student_grade = 'D+' THEN 1.3
        WHEN student_grade = 'D' THEN 1.0
        WHEN student_grade = 'E' THEN 0.0
        ELSE 0.0
    END) / SUM(credits), 2) AS CGPA
FROM course_credits
WHERE course_code != 'ENG1222'
GROUP BY student_id;




CREATE VIEW CA_Eligibility_new AS 
SELECT
        stu_id,
        c_code,
        final_CA_marks,
        

        CASE
             
             WHEN c_code = "ICT2113" AND  final_CA_marks >= (15 * 0.3) THEN "Pass"
             WHEN c_code = "ICT2122" AND  final_CA_marks >= (20 * 0.4) THEN "Pass"
             WHEN c_code = "ICT2133" AND  final_CA_marks >= (15 * 0.3) THEN "Pass"
             WHEN c_code = "ICT2142" AND  final_CA_marks >= (20 * 0.4) THEN "Pass"
             WHEN c_code = "ICT2152" AND  final_CA_marks >= (15 * 0.3) THEN "Pass"
             ELSE "Fail"   
        END
        AS CA_eligibility
FROM 
        final_CA_marks; 



CREATE VIEW attendance_eligibility AS
SELECT 
    username,
    c_code,
    c_type,
    SUM(CASE WHEN at_state = 'Present' OR at_state = 'Medical' THEN c_hours ELSE 0 END) * 100.0 / SUM(c_hours) AS Percentage,
    CASE 
        WHEN (SUM(CASE WHEN at_state = 'Present' OR at_state = 'Medical' THEN c_hours ELSE 0 END) * 100.0 / SUM(c_hours)) >= 80
        THEN 'Eligible'
        ELSE 'Not_Eligible' 
    END AS Eligibility
FROM 
    attendance_detail
GROUP BY 
    username, c_code, c_type;


CREATE VIEW final_eligibility AS
SELECT 
    ca.stu_id,
    ca.c_code,
    
    CASE
        WHEN ca.CA_eligibility = 'Pass' AND atd.Eligibility = 'Eligible' THEN 'Eligible'
        ELSE 'Not_Eligible'
    END AS Final_Eligibility
FROM 
    CA_Eligibility_new ca
JOIN 
    attendance_eligibility atd
ON 
    ca.c_code = atd.c_code
AND 
    ca.stu_id = atd.username;
